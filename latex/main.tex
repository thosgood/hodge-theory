\begin{abstract}
  This is an English translation of \textbf{P. Deligne}'s three papers on Hodge theory:

  \begin{itemize}
    \item
      "Théorie de Hodge I",
      \emph{Actes du Congrès intern. math.} \textbf{1} (1970) pp. 425–430.
      \\\url{publications.ias.edu/node/359}
    
    \item
      "Théorie de Hodge II",
      \emph{Pub. Math. de l'IHÉS} \textbf{40} (1971) pp. 5–58.
      \\\url{publications.ias.edu/node/361}
    
    \item
      "Théorie de Hodge III",
      \emph{Pub. Math. de l'IHÉS} \textbf{44} (1974) pp. 5–77.
      \\\url{publications.ias.edu/node/373}
    
  \end{itemize}

  Corrections and comments welcome.
\end{abstract}
